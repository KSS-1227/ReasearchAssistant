def smart_rag_answer(
    question: str,
    coordinator,
    llm_call_fn,
    has_documents: bool = True,
    local_threshold: float = LOCAL_CONFIDENCE_THRESHOLD,
) -> SmartRagResult:
    """
    Implements the exact spec:
      1. Always try local FAISS first.
      2. best_score >= threshold -> local only, no web call.
      3. best_score <  threshold -> LLM scope check:
           in scope     -> web search, fuse local+web, answer from both.
           out of scope -> decline politely, never call web_search.
      4. Final answer always tags its source: local | web | both | declined.
    """
    # Step 1: local retrieval (always attempted)
    local = retrieve_local(coordinator, question)

    # Step 2: threshold check
    if local.has_results and local.best_score >= local_threshold:
        logger.info(
            "smart_rag_answer: best_score=%.3f >= threshold=%.3f -> LOCAL ONLY",
            local.best_score, local_threshold,
        )
        synthesis = coordinator.research_query(question)
        synthesis["source_attribution"] = "local"
        return SmartRagResult(
            answer_source="local",
            declined=False,
            decline_reason=None,
            local=local,
            web=WebRetrieval(),
            synthesis=synthesis,
        )

    # Step 3: below threshold -> scope check before any web call
    logger.info(
        "smart_rag_answer: best_score=%.3f < threshold=%.3f -> running scope check",
        local.best_score, local_threshold,
    )
    scope = check_scope(llm_call_fn, question, has_documents, local.best_score)

    if not scope.in_scope:
        logger.info("smart_rag_answer: query OUT OF SCOPE (%s) - declining, no web call", scope.reasoning)
        return SmartRagResult(
            answer_source="declined",
            declined=True,
            decline_reason=scope.reasoning,
            local=local,
            web=WebRetrieval(),
            synthesis=None,
        )

    # Step 4: in scope -> web search, fuse with whatever local context exists
    web = retrieve_web(question)

    if not local.has_results and not web.has_results:
        # Nothing from either source - decline rather than hallucinate
        return SmartRagResult(
            answer_source="declined",
            declined=True,
            decline_reason="No relevant information found in documents or the web.",
            local=local,
            web=web,
            synthesis=None,
        )

    fused_question = question
    if web.context:
        fused_question = (
            f"{question}\n\n"
            f"[Supplementary web research - use alongside any local document "
            f"context to answer]\n{web.context}"
        )

    synthesis = coordinator.research_query(fused_question)

    source = "both" if (local.has_results and web.has_results) else (
        "local" if local.has_results else "web"
    )
    synthesis["source_attribution"] = source
    synthesis["web_sources"] = web.sources

    logger.info("smart_rag_answer: answered from source='%s'", source)

    return SmartRagResult(
        answer_source=source,
        declined=False,
        decline_reason=None,
        local=local,
        web=web,
        synthesis=synthesis,
    )
